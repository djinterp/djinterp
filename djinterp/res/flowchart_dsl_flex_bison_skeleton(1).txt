// ============================================================================
// FLOWCHART DSL FLEX/BISON SKELETON
// ============================================================================
// This is a practical conversion of the ABNF grammar into a Bison/Flex starting
// point. It keeps most grammar structure, but intentionally parses expressions
// as opaque expression text in several places because a complete expression
// grammar was not included in the ABNF.
//
// Suggested files:
//   flowchart_parser.y
//   flowchart_lexer.l
//   flowchart_ast.hpp
//   flowchart_driver.hpp / .cpp
// ============================================================================


// ============================================================================
// FILE: flowchart_parser.y
// ============================================================================

%language "c++"
%require "3.8"
%defines "flowchart_parser.hpp"
%output  "flowchart_parser.cpp"

%define api.namespace {flowchart}
%define api.parser.class {parser}
%define api.value.type variant
%define api.token.constructor
%define parse.error detailed
%define parse.lac full

%locations

%code requires {
    #include <memory>
    #include <string>
    #include <vector>

    namespace flowchart
    {
        class driver;

        struct ast_node
        {
            virtual ~ast_node() = default;
        };

        struct document : ast_node
        {
            std::vector<std::unique_ptr<ast_node>> elements;
        };
    }
}

%parse-param { flowchart::driver& drv }
%lex-param   { flowchart::driver& drv }

%code {
    #include "flowchart_driver.hpp"
    static flowchart::parser::symbol_type yylex(flowchart::driver& drv);
}

// ----------------------------------------------------------------------------

// ============================================================================
// Document
// ============================================================================

flowchart_document
    : flowchart_header flowchart_element_list
      {
          $$ = std::make_unique<flowchart::document>();
          drv.set_document(std::move($$));
      }
    ;

flowchart_header
    : dimension_spec opt_time_spec opt_default_styles
    ;

flowchart_element_list
    : /* empty */
    | flowchart_element_list flowchart_element
    ;

flowchart_element
    : node_definition
    | edge_definition
    | connector_definition
    | group_definition
    | annotation_definition
    | layer_definition
    ;

// ============================================================================
// Dimension / time / defaults
// ============================================================================

dimension_spec
    : KW_DIMENSION ':' dimension_value
    | KW_DIM ':' dim_short_value
    ;

dimension_value
    : KW_2D
    | KW_3D
    ;

dim_short_value
    : integer
      {
          /* semantic check: must be 2 or 3 */
      }
    ;

opt_time_spec
    : /* empty */
    | time_spec
    ;

time_spec
    : KW_TIME ':' time_mode
    | KW_T ':' time_range
    ;

time_mode
    : KW_STATIC
    | KW_ANIMATED
    | KW_TIMELINE
    ;

time_range
    : expression DOTDOT expression
    | '[' expression ',' expression ']'
    ;

opt_default_styles
    : /* empty */
    | default_styles
    ;

default_styles
    : KW_DEFAULTS '{' style_property_list '}'
    ;

// ============================================================================
// Nodes / shapes
// ============================================================================

node_definition
    : KW_NODE IDENTIFIER opt_node_type position_spec opt_style_block opt_content_block
      { $$ = nullptr; }
    ;

opt_node_type
    : /* empty */
    | ':' shape_spec
    ;

shape_spec
    : basic_shape
    | solid_shape
    | custom_shape
    ;

basic_shape
    : KW_RECTANGLE | KW_RECT | KW_BOX | KW_SQUARE
    | KW_CIRCLE | KW_ELLIPSE | KW_OVAL
    | KW_DIAMOND | KW_RHOMBUS | KW_PARALLELOGRAM | KW_TRAPEZOID
    | KW_HEXAGON | KW_OCTAGON | KW_PENTAGON
    | KW_TRIANGLE | KW_TRIANGLE_UP | KW_TRIANGLE_DOWN | KW_TRIANGLE_LEFT | KW_TRIANGLE_RIGHT
    | KW_STAR opt_integer_call
    | KW_ARROW | KW_ARROW_LEFT | KW_ARROW_RIGHT | KW_ARROW_UP | KW_ARROW_DOWN
    | KW_CLOUD | KW_CYLINDER | KW_DOCUMENT | KW_TAPE | KW_STORED_DATA
    | KW_MANUAL_INPUT | KW_MANUAL_OPERATION | KW_PROCESS | KW_DECISION | KW_DATA
    | KW_TERMINATOR | KW_START | KW_END
    | KW_PREDEFINED_PROCESS | KW_SUBPROCESS | KW_INTERNAL_STORAGE | KW_DELAY | KW_PREPARATION
    | KW_OFF_PAGE_CONNECTOR | KW_OR_GATE | KW_SUMMING_JUNCTION
    | KW_DATABASE | KW_DB | KW_QUEUE | KW_COLLATE | KW_SORT | KW_MERGE | KW_EXTRACT | KW_DISPLAY
    ;

solid_shape
    : KW_CUBE | KW_BOX3D | KW_SPHERE | KW_CONE | KW_PYRAMID
    | KW_PRISM opt_integer_call
    | KW_TORUS | KW_CYLINDER3D | KW_TETRAHEDRON | KW_OCTAHEDRON
    | KW_DODECAHEDRON | KW_ICOSAHEDRON | KW_CAPSULE | KW_ELLIPSOID
    ;

opt_integer_call
    : /* empty */
    | '(' integer ')'
    ;

custom_shape
    : KW_CUSTOM '{' shape_definition '}'
    ;

shape_definition
    : parametric_shape
    | implicit_shape
    | point_list_shape
    ;

parametric_shape
    : IDENTIFIER ':' expression ',' IDENTIFIER ':' expression opt_z_expression ',' parameter_range
      {
          /* semantic check: first two identifiers should be x and y; optional z */
      }
    ;

opt_z_expression
    : /* empty */
    | ',' IDENTIFIER ':' expression
      {
          /* semantic check: identifier should be z */
      }
    ;

implicit_shape
    : KW_EQUATION ':' expression opt_comparison expression
    ;

opt_comparison
    : /* empty */
    | '<'
    | '>'
    | '='
    | LE
    | GE
    ;

point_list_shape
    : KW_POINTS ':' point_list opt_closed_property
    ;

opt_closed_property
    : /* empty */
    | ',' KW_CLOSED ':' boolean
    ;

point_list
    : '[' point_items ']'
    ;

point_items
    : point
    | point_items ',' point
    ;

parameter_range
    : IDENTIFIER IN_SYMBOL '[' expression ',' expression ']'
    | IDENTIFIER KW_IN range_expression
    ;

range_expression
    : expression DOTDOT expression
    | '[' expression ',' expression ']'
    ;

// ============================================================================
// Position
// ============================================================================

position_spec
    : KW_AT position opt_orientation
    ;

position
    : coordinate_2d
    | coordinate_3d
    | relative_position
    | grid_position
    | mathematical_position
    ;

coordinate_2d
    : '(' expression ',' expression ')'
    ;

coordinate_3d
    : '(' expression ',' expression ',' expression ')'
    ;

relative_position
    : anchor_point opt_offset
    ;

anchor_point
    : IDENTIFIER '.' anchor_name
    ;

anchor_name
    : KW_CENTER | KW_C
    | KW_NORTH | KW_N | KW_TOP
    | KW_SOUTH | KW_S | KW_BOTTOM
    | KW_EAST | KW_E | KW_RIGHT
    | KW_WEST | KW_W | KW_LEFT
    | KW_NORTHEAST | KW_NE | KW_TOP_RIGHT
    | KW_NORTHWEST | KW_NW | KW_TOP_LEFT
    | KW_SOUTHEAST | KW_SE | KW_BOTTOM_RIGHT
    | KW_SOUTHWEST | KW_SW | KW_BOTTOM_LEFT
    ;

opt_offset
    : /* empty */
    | '+' vector_expression
    | '-' vector_expression
    ;

vector_expression
    : point
    | expression
    ;

grid_position
    : '[' integer ',' integer opt_grid_z ']'
    ;

opt_grid_z
    : /* empty */
    | ',' integer
    ;

mathematical_position
    : KW_F_UNIT '(' expression opt_arg_expr opt_arg_expr ')'
    | KW_POLAR '(' expression ',' expression ')'
    | KW_CYLINDRICAL '(' expression ',' expression ',' expression ')'
    | KW_SPHERICAL '(' expression ',' expression ',' expression ')'
    ;

opt_arg_expr
    : /* empty */
    | ',' expression
    ;

opt_orientation
    : /* empty */
    | orientation
    ;

orientation
    : KW_ROTATE ':' angle
    | KW_PITCH ':' angle ',' KW_YAW ':' angle ',' KW_ROLL ':' angle
    ;

angle
    : expression opt_angle_unit { $$ = $1; }
    ;

opt_angle_unit
    : /* empty */
    | angle_unit
    ;

angle_unit
    : KW_DEG | DEG_SYMBOL | KW_RAD | KW_GRAD | KW_TURN
    ;

// ============================================================================
// Edges / waypoints
// ============================================================================

edge_definition
    : KW_EDGE IDENTIFIER edge_spec opt_edge_style opt_edge_labels
      { $$ = nullptr; }
    ;

edge_spec
    : KW_FROM node_reference KW_TO node_reference opt_waypoints
    ;

node_reference
    : IDENTIFIER opt_anchor_suffix
    ;

opt_anchor_suffix
    : /* empty */
    | '.' anchor_name
    ;

opt_waypoints
    : /* empty */
    | KW_VIA waypoint_list
    ;

waypoint_list
    : waypoint
    | waypoint_list ',' waypoint
    ;

waypoint
    : position
    | KW_MANHATTAN
    | KW_ORTHOGONAL
    | KW_SMOOTH
    | KW_CURVED
    | KW_BEZIER '(' control_points ')'
    | KW_SPLINE
    ;

control_points
    : point
    | control_points ',' point
    ;

point
    : '(' expression ',' expression opt_point_z ')'
    ;

opt_point_z
    : /* empty */
    | ',' expression
    ;

opt_edge_style
    : /* empty */
    | edge_style
    ;

edge_style
    : KW_STYLE '{' edge_style_property_list '}'
    ;

edge_style_property_list
    : /* empty */
    | edge_style_property_list edge_style_property
    ;

edge_style_property
    : line_property
    | arrow_property
    | color_property
    | animation_property
    ;

// ============================================================================
// Line / arrow / color properties
// ============================================================================

line_property
    : line_type
    | line_width
    | line_pattern
    | line_cap
    | line_join
    ;

line_type
    : KW_TYPE ':' line_style_value
    ;

line_style_value
    : KW_SOLID
    | KW_DASHED
    | KW_DOTTED
    | KW_DASH_DOT
    | KW_DASH_DOT_DOT
    | KW_CUSTOM '[' number_list ']'
    ;

line_width
    : KW_WIDTH ':' measurement
    | KW_THICKNESS ':' measurement
    | KW_WEIGHT ':' line_weight_value
    ;

line_weight_value
    : KW_THIN
    | KW_MEDIUM
    | KW_THICK
    | measurement
    ;

line_pattern
    : KW_PATTERN ':' line_style_value
    ;

line_cap
    : KW_CAP ':' line_cap_value
    ;

line_cap_value
    : KW_BUTT
    | KW_ROUND
    | KW_SQUARE
    ;

line_join
    : KW_JOIN ':' line_join_value
    ;

line_join_value
    : KW_MITER
    | KW_ROUND
    | KW_BEVEL
    ;

arrow_property
    : KW_ARROW_START ':' arrow_style
    | KW_ARROW_END ':' arrow_style
    | KW_ARROW ':' arrow_style
    | KW_ARROW_SIZE ':' measurement
    ;

arrow_style
    : KW_NONE
    | KW_ARROW
    | '>'
    | KW_TRIANGLE
    | ARROW_TRIANGLE
    | KW_OPEN
    | ARROW_OPEN
    | KW_DIAMOND
    | ARROW_DIAMOND
    | KW_FILLED_DIAMOND
    | ARROW_FILLED_DIAMOND
    | KW_CIRCLE
    | ARROW_CIRCLE
    | KW_FILLED_CIRCLE
    | ARROW_FILLED_CIRCLE
    | KW_SQUARE
    | ARROW_SQUARE
    | KW_FILLED_SQUARE
    | ARROW_FILLED_SQUARE
    | KW_BAR
    | KW_PERPENDICULAR
    | ARROW_PERPENDICULAR
    | KW_CROW_FOOT
    | KW_FORK
    | KW_MANY
    | KW_ONE
    | KW_ZERO_OR_MORE
    | KW_ONE_OR_MORE
    | KW_ZERO_OR_ONE
    | KW_CUSTOM '{' shape_definition '}'
    ;

color_property
    : stroke_color
    | fill_color
    | gradient_fill
    | opacity_property
    ;

stroke_color
    : KW_STROKE ':' color_value
    | KW_BORDER_COLOR ':' color_value
    | KW_LINE_COLOR ':' color_value
    ;

fill_color
    : KW_FILL ':' color_value
    | KW_BACKGROUND ':' color_value
    | KW_BG ':' color_value
    ;

gradient_fill
    : KW_GRADIENT ':' gradient_spec
    ;

gradient_spec
    : linear_gradient
    | radial_gradient
    | conic_gradient
    ;

linear_gradient
    : KW_LINEAR '(' angle ',' color_stop_list ')'
    | KW_LINEAR '(' direction ',' color_stop_list ')'
    ;

radial_gradient
    : KW_RADIAL '(' position ',' color_stop_list ')'
    | KW_RADIAL '(' radial_shape ',' color_stop_list ')'
    ;

radial_shape
    : KW_CIRCLE
    | KW_ELLIPSE
    ;

conic_gradient
    : KW_CONIC '(' angle ',' position ',' color_stop_list ')'
    ;

direction
    : KW_TO direction_value
    ;

direction_value
    : KW_TOP | KW_BOTTOM | KW_LEFT | KW_RIGHT
    | KW_TOP_LEFT | KW_TOP_RIGHT | KW_BOTTOM_LEFT | KW_BOTTOM_RIGHT
    ;

color_stop_list
    : color_stop
    | color_stop_list ',' color_stop
    ;

color_stop
    : color_value opt_percentage
    ;

opt_percentage
    : /* empty */
    | percentage
    ;

color_value
    : HEX_COLOR
    | rgb_color
    | rgba_color
    | hsl_color
    | hsla_color
    | named_color
    | IDENTIFIER
    ;

rgb_color
    : KW_RGB '(' integer ',' integer ',' integer ')'
    ;

rgba_color
    : KW_RGBA '(' integer ',' integer ',' integer ',' number ')'
    ;

hsl_color
    : KW_HSL '(' number ',' percentage ',' percentage ')'
    ;

hsla_color
    : KW_HSLA '(' number ',' percentage ',' percentage ',' number ')'
    ;

named_color
    : KW_RED | KW_GREEN | KW_BLUE | KW_YELLOW | KW_ORANGE | KW_PURPLE | KW_PINK
    | KW_BLACK | KW_WHITE | KW_GRAY | KW_GREY | KW_CYAN | KW_MAGENTA
    | KW_BROWN | KW_LIME | KW_NAVY | KW_TEAL | KW_OLIVE | KW_MAROON
    | KW_SILVER | KW_GOLD | KW_TRANSPARENT | KW_AQUA | KW_FUCHSIA
    | KW_INDIGO | KW_VIOLET | KW_SALMON | KW_CORAL | KW_CRIMSON
    ;

opacity_property
    : KW_OPACITY ':' number
    | KW_ALPHA ':' number
    ;

// ============================================================================
// Style properties
// ============================================================================

style_property_list
    : /* empty */
    | style_property_list style_property
    ;

style_property
    : color_property
    | border_property
    | spacing_property
    | dimension_property
    | shadow_property
    | transform_property
    ;

border_property
    : KW_BORDER_WIDTH ':' measurement
    | KW_BORDER ':' measurement opt_line_style_value opt_color_value
    | KW_BORDER_STYLE ':' line_style_value
    | KW_BORDER_COLOR ':' color_value

// ============================================================================
// IMPORTANT FIXUP NOTES
// ============================================================================
// 1. This skeleton deliberately uses EXPR as an opaque token, but the lexer above
//    does not yet produce EXPR. You have two sane options:
//
//    A. Define a real expression grammar in Bison:
//          expression : expression '+' expression
//                     | expression '*' expression
//                     | IDENTIFIER
//                     | NUMBER
//                     | '(' expression ')'
//                     ;
//
//    B. Lex expressions in context until a delimiter. That requires lexer start
//       conditions because expression delimiters differ by context.
//
//    For a serious DSL, option A is better.
//
// 2. The ABNF has many bare words that can be both keywords and identifiers,
//    such as "w", "h", "d", "s", "f", "c". In a scannerless parser this is
//    easy; in Flex/Bison it can cause conflicts. Better design:
//       - reserve only true syntax introducers as keywords;
//       - parse enum-like property values as IDENTIFIER;
//       - validate them semantically.
//
// 3. Several tokens referenced in parser.y still need keyword rules if you keep
//    the fully tokenized style: KW_RED, KW_DOT, KW_ABOVE, KW_BELOW, KW_EASE,
//    KW_EASE_IN, KW_EASE_OUT, KW_EASE_IN_OUT, KW_CUBIC_BEZIER, KW_TO, etc.
//    You can either add those tokens/rules or simplify these sets to IDENTIFIER
//    plus semantic validation.
//
// 4. Recommended next pass: collapse most value words into IDENTIFIER. For
//    example, instead of keyword tokens for every shape, do this:
//
//        basic_shape : IDENTIFIER opt_integer_call;
//
//    Then validate that IDENTIFIER is one of the allowed shape names.
//
// 5. The grammar currently allows style_property -> spacing_value, which means a
//    raw measurement can appear directly in a style block. That follows the ABNF,
//    but it may make error messages worse.
