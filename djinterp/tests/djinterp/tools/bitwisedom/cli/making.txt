# new, add and change: rows made and defined as a bitfield's rows are, and
# refused as the window refuses them
$ bitwisedom new --width=16 --title=regs -o s.json
$ bitwisedom add s.json FLAG_A --bit=1 -v
! bitwisedom: note: FLAG_A added, at bit 1
$ bitwisedom add s.json FLAG_B --bit -v
! bitwisedom: note: FLAG_B added, at bit 0
$ bitwisedom add s.json MODE --lsb=4 --width=3 --note=the_mode -v
! bitwisedom: note: MODE added, at bits 4..6
$ bitwisedom add s.json MODE_FAST --field=MODE --number=2 -v
! bitwisedom: note: MODE_FAST added
$ bitwisedom add s.json BOTH --expr='FLAG_A | FLAG_B' -v
! bitwisedom: note: BOTH added
$ bitwisedom add s.json SIXTEEN --value=0x10 -v
! bitwisedom: note: SIXTEEN added
$ bitwisedom add s.json SQUARE --body='((x) * (x))' --params=x -v
! bitwisedom: note: SQUARE added
$ bitwisedom add s.json Ctrl --group --prefix=CTRL_ -v
! bitwisedom: note: Ctrl added
$ bitwisedom add s.json EN --in=Ctrl --bit=8 -v
! bitwisedom: note: CTRL_EN added, at bit 8
